P=2; OWNER=Jess-Rivera

FIELD=$(gh project field-list $P --owner $OWNER --format json --jq '.fields[] | select(.name=="Status") | .id')

if [ -z "$PROJECT" ] || [ -z "$FIELD" ] || [ -z "$OPTION" ] || [ -z "$ITEM" ]; then echo "lookup failed: project=$PROJECT field=$FIELD option=$OPTION item=$ITEM"; else gh project item-edit --id "$ITEM" --project-id "$PROJECT" --field-id "$FIELD" --single-select-option-id "$OPTION" && echo "#39 moved to Needs Playtest"; fi
